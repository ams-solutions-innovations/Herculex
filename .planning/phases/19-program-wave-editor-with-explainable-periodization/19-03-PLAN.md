<!-- Periodization phase-boundary source of truth for Block's week numbers.
     Source: lib/features/programs/domain/periodization.dart:121-156.
     For an 8-week block: accWeeks = ceil(8*0.4)=4 (weeks 1-4 accumulation),
     transCap = 8-4-1=3, transWeeks = ceil(8*0.4).clamp(0,3)=3 (weeks 5-7 transmutation),
     remaining week 8 = realization. Guide week numbers for 'block' MUST match this exactly. -->
```dart
static List<WeekPrescription> _block(int weeks) {
  final accWeeks = (weeks * 0.4).ceil().clamp(1, weeks);
  final transCap = weeks - accWeeks - (weeks >= 3 ? 1 : 0);
  final transWeeks = (weeks * 0.4).ceil().clamp(0, transCap < 0 ? 0 : transCap);
  // w < accWeeks -> accumulation; w < accWeeks+transWeeks -> transmutation; else realization
}
```
