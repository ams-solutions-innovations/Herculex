<!-- SessionSegment enum shape to create — mirrors SetType.fromId / SlotRole.fromId
     conventions, but returns null (not a default value) since null = "no segment"
     is a real, common state for every non-CrossFit/GPP row. -->
enum SessionSegment {
  warmup('warmup'),
  skill('skill'),
  strength('strength'),
  metcon('metcon'),
  cooldown('cooldown');

  const SessionSegment(this.id);
  final String id;

  static SessionSegment? fromId(String? id) { /* null-safe lookup, no default */ }
}

<!-- Shared public descriptor type crossfit_program_planner.dart (21-04) and
     gpp_program_planner.dart (21-05) both return in Wave 2 — lives here, not
     in either planner file, so the two Wave 2 plans stay file-independent
     and can run in parallel. Field shape mirrors smart_program_planner.dart's
     existing private _SlotNeed (pattern/muscle/role/preferredSlugs), extended
     with the segment tag and metcon-only fields. -->
class CrossfitSlotNeed {
  const CrossfitSlotNeed({
    this.pattern,
    this.muscle,
    required this.role,
    this.preferredSlugs = const {},
    required this.segment,
    this.metconGroupKey,
    this.metconFormat,
    this.metconCapSeconds,
    this.metconMinutes,
  });
  final String? pattern;
  final String? muscle;
  final SlotRole role;
  final Set<String> preferredSlugs;
  final SessionSegment segment;
  final String? metconGroupKey;
  final SetType? metconFormat;
  final int? metconCapSeconds;
  final int? metconMinutes;
}
