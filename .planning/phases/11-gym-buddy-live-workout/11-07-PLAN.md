  class BuddySlot {
    final String slotId;
    final int? workoutExerciseId;      // null => placeholder
    final String? unresolvedUuid;
    final String? unresolvedSlug;
    final String? placeholderLabel;
    final int orderIndex;
    bool get isPlaceholder => workoutExerciseId == null;
  }

  class BuddySlotStore {
    BuddySlotStore(AppDatabase db, String buddySessionId);
    Future<BuddySlot?> bySlotId(String slotId);
    Future<BuddySlot?> byWorkoutExerciseId(int id);   // 11-08 needs the reverse lookup
    Future<List<BuddySlot>> all();                    // ordered by orderIndex
    Future<void> upsert(BuddySlot slot);
    Future<void> unlink(String slotId);               // drops the slot row, keeps the exercise
    Future<void> deleteSlot(String slotId);
  }
