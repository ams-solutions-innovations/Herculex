lib/features/buddy/domain/buddy_event.dart — the wire contract. No `scope` field, ever.

  enum BuddyEventKind { add, remove, reorder, replace, sessionEnded }
    // wireName: 'add' | 'remove' | 'reorder' | 'replace' | 'session_ended'
    // BuddyEventKind.fromWire(String) -> BuddyEventKind?   (null on unknown kind)

  class BuddyExerciseRef            // exactly one of the two is non-null
    final String? uuid;             // sync_uuid, for a custom catalogue row
    final String? slug;             // stable natural key, for a seeded row
    Map<String, dynamic> toJson();  // {'uuid': ..., 'slug': ...}
    factory BuddyExerciseRef.fromJson(Map<String, dynamic>);

  class BuddyEvent
    final String buddySessionId;
    final int seq;
    final String actorUserId;
    final BuddyEventKind kind;
    final Map<String, dynamic> payload;
    factory BuddyEvent.fromBroadcast(Map<String, dynamic> payload);  // seq/kind/actor/payload
    factory BuddyEvent.fromLogRow(Map<String, dynamic> row);         // buddy_session_id/seq/actor_user_id/kind/payload

  class BuddyAddPayload     { String slotId; BuddyExerciseRef ref; String? afterSlotId; String? equipmentVariant; }
  class BuddyRemovePayload  { String slotId; }
  class BuddyReorderPayload { List<String> order; }   // FULL absolute ordering, never a delta
  class BuddyReplacePayload { String slotId; BuddyExerciseRef ref; }
  class BuddySessionEndedPayload { String endedBy; }
  // each with toJson() / fromJson(Map<String, dynamic>)

lib/features/buddy/domain/buddy_scope.dart — NOT part of the transport module.

  enum BuddyScope { both, mine }
  enum BuddyActionKind { add, remove, reorder, replace }
  abstract final class BuddyScopeDefaults {
    static BuddyScope forAction(BuddyActionKind kind);
    // add / reorder / replace -> BuddyScope.both
    // remove                  -> BuddyScope.mine
  }
